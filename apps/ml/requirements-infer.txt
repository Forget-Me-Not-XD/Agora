# Pi inference dependencies — do NOT add tensorflow here.
# Full TensorFlow (~500 MB) will not install cleanly on Raspberry Pi OS Bookworm.

# tflite-runtime vs ai-edge-litert — the tradeoff:
#   tflite-runtime  uses: import tflite_runtime.interpreter  (deprecated after 2.14.0)
#   ai-edge-litert  uses: import ai_edge_litert.interpreter  (Google's replacement)
# model.tflite is exported by train.py using TensorFlow 2.15.1, whose converter
# emits op versions (e.g. FULLY_CONNECTED v12) that tflite-runtime 2.14.0's
# interpreter doesn't recognise. ai-edge-litert supports these newer op
# versions, so predict.py now imports ai_edge_litert.interpreter.
ai-edge-litert>=1.0.1

# Array operations — building and reshaping the (1, 5, 4) input tensor.
numpy>=1.24.0

# Required to deserialise scaler.pkl.
# pickle.load() must reconstruct a MinMaxScaler object, which means Python
# needs sklearn.preprocessing available at load time — even though we never
# call any sklearn functions directly in predict.py.
scikit-learn>=1.3.0
